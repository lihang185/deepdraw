cmake_minimum_required(VERSION 3.19)
project(EducationSketchUp LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SketchUp SDK requires Windows x64")
endif()
find_package(Python 3.8 EXACT REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sketchup")
foreach(module suapi suapi_extra skp_exporter)
  pybind11_add_module(${module} ${module}/bindings.cpp)
  target_include_directories(${module} PRIVATE "${SDK}/include")
  target_link_libraries(${module} PRIVATE "${SDK}/lib/SketchUpAPI.lib")
  target_compile_options(${module} PRIVATE /utf-8 /EHsc)
  install(TARGETS ${module} LIBRARY DESTINATION . RUNTIME DESTINATION .)
endforeach()
target_sources(suapi PRIVATE suapi/model.cpp suapi/builders.cpp)
target_sources(skp_exporter PRIVATE skp_exporter/SkpExporter.cpp)
set(ENGINE_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/native/cp38")
target_include_directories(skp_exporter PRIVATE "${ENGINE_NATIVE}")
target_link_libraries(skp_exporter PRIVATE "${ENGINE_NATIVE}/editor2.cp38-win_amd64.lib")
